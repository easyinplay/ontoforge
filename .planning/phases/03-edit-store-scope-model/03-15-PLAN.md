## Phase Goal

```rust
// crates/ontoforge-storage/src/history.rs（读侧，新增）
#[derive(Debug, Clone, Copy, PartialEq, Eq)] pub enum CallerKind { Ordinary, Admin }
pub struct HistoryRequest<'a> { pub type_name: &'a ObjectTypeName, pub pk: &'a PropertyValue,
    pub scope: Option<&'a TypeScope>,                     // None = 调用者对该 type 无授权
    pub attrs: &'a Attrs, pub current_row: Option<&'a dyn RowValues>,   // None = 对象不存在（调用方已叠加 source）
    pub caller: CallerKind, pub limit: u32, pub before_seq: Option<i64> }
#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct HistoryEntry { pub seq: i64, pub event_id: String, pub actor: String, pub action: String, pub committed_at: String,
    pub op: EditOp, pub version: i64,
    pub values: BTreeMap<PropertyName, serde_json::Value>,   // 可见列在本次 set 的新值
    pub unset: BTreeSet<PropertyName>,                       // 可见列在本次被 unset
    pub masked_changes: BTreeSet<PropertyName>,              // 不可见但被改动的列：只标记、无值
    #[serde(skip_serializing_if = "Option::is_none")] pub params: Option<serde_json::Value> }
#[derive(Debug, thiserror::Error)] pub enum HistoryError { #[error("object not found")] NotFound, #[error(transparent)] Store(StoreError) }
impl EditStore { pub async fn history(&self, req: &HistoryRequest<'_>) -> Result<Vec<HistoryEntry>, HistoryError>; }
```
